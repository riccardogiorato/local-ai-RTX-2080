    <script>
        /**
         * Lava Lamp Simulation Logic
         * Uses HTML5 Canvas for metaball rendering.
         */
        const canvas = document.getElementById('wax-layer');
        const ctx = canvas.getContext('2d');
        
        // Configuration
        const config = {
            blobCount: 15,
            blobRadius: 40,
            riseSpeed: 0.5,
            fallSpeed: 0.8,
            turbulence: 0.8, // How much they wiggle
            glowStrength: 30
        };

        // State
        let blobs = [];
        let particles = [];
        let time = 0;

        // Resize handling
        function resize() {
            const container = document.getElementById('lava-container');
            canvas.width = container.offsetWidth;
            canvas.height = container.offsetHeight;
        }

        /**
         * Class representing a single wax blob
         */
        class Blob {
            constructor() {
                this.reset(true);
            }

            reset(initial = false) {
                this.x = Math.random() * canvas.width;
                this.y = Math.random() * canvas.height;
                this.radius = config.blobRadius * (0.8 + Math.random() * 0.4);
                
                // Physics
                this.vy = initial ? -Math.random() * 2 : -0.5; // Initial upward pop
                this.vy += (Math.random() - 0.5) * 0.5;
                
                this.vx = (Math.random() - 0.5) * 1; // Slight horizontal drift
                this.wobble = Math.random() * Math.PI * 2;
                this.wobbleSpeed = 0.02 + Math.random() * 0.03;
                
                // Color (Warm yellows/oranges)
                this.hue = 30 + Math.random() * 40; 
                this.sat = 70 + Math.random() * 30;
                this.light = 50 + Math.random() * 30;
            }

            update() {
                // Physics: Gravity
                this.vy += config.fallSpeed;
                this.y += this.vy;
                this.x += this.vx;

                // Wobble movement
                this.x += Math.sin(time * this.wobbleSpeed + this.wobble) * config.turbulence;

                // Boundaries
                // Top (Neck area)
                if (this.y < 150) {
                    this.y = 150;
                    this.vy = -Math.random() * 2; // Bounce up
                }
                // Bottom (Base)
                if (this.y > 450) {
                    this.y = 450;
                    this.vy = -Math.random() * 1.5; // Bounce up
                }
                // Walls
                if (this.x < 50) {
                    this.x = 50;
                    this.vx *= -1;
                }
                if (this.x > canvas.width - 50) {
                    this.x = canvas.width - 50;
                    this.vx *= -1;
                }
            }

            draw(ctx) {
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fillStyle = `hsla(${this.hue}, ${this.sat}%, ${this.light}%, 0.9)`;
                ctx.fill();
                
                // Inner highlight (specular)
                ctx.beginPath();
                ctx.arc(this.x - this.radius * 0.3, this.y - this.radius * 0.3, this.radius * 0.2, 0, Math.PI * 2);
                ctx.fillStyle = `hsla(${this.hue}, ${this.sat}%, 90%, 0.4)`;
                ctx.fill();
            }
        }

        // Initialize Blobs
        for (let i = 0; i < config.blobCount; i++) {
            blobs.push(new Blob());
        }

        /**
         * Metaball Rendering Logic
         * Uses a threshold-based approach to create the "gooey" effect.
         */
        function drawMetaballs() {
            // Clear canvas
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Create an image data buffer to store "heat" data
            const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
            const data = imgData.data;

            // 1. Fill background with black (or very dark)
            for (let i = 0; i < data.length; i += 4) {
                data[i] = 0;     // R
                data[i+1] = 0;   // G
                data[i+2] = 0;   // B
                data[i+3] = 1;   // Alpha
            }

            // 2. Draw blobs and accumulate heat
            for (let blob of blobs) {
                // We draw the blob with a semi-transparent white color.
                // This adds to the "heat" value in the buffer.
                ctx.beginPath();
                ctx.arc(blob.x, blob.y, blob.radius, 0, Math.PI * 2);
                ctx.fillStyle = `rgba(255, 255, 255, 0.15)`; // Low opacity for accumulation
                ctx.fill();
            }

            // 3. Reconstruct the image
            // We need to map the accumulated heat back to the blob colors.
            // Instead of modifying the buffer directly (which is tricky with ImageData),
            // we'll iterate over the buffer and check neighbors.
            
            // Note: Doing this efficiently in pure JS without libraries is heavy.
            // We will use a simpler thresholding approach on the buffer for this demo.
            // We will draw the blobs again but masked by the buffer threshold.
            
            // Actually, the "Metaball" trick in vanilla JS is:
            // 1. Draw blobs with low alpha (say 0.1)
            // 2. Create a buffer.
            // 3. Iterate buffer. If pixel > threshold, it's a blob.
            // 4. Draw a mask based on that.
            
            // Let's do the standard "Threshold" method:
            // We will draw the blobs again, but this time we don't use ImageData for the final 
            // shape, we use a logic check.
            
            // Reset data for the "mask" pass
            const maskData = ctx.createImageData(canvas.width, canvas.height);
            const mask = new Uint32Array(maskData.data.buffer);
            
            // Fill with transparent
            for(let i=0; i<mask.length; i++) mask[i] = 0xFFFFFFFF;

            // Helper: Check if a pixel is "hot" enough
            // We check a pixel and its 4 neighbors. If ANY are hot, this pixel is hot.
            function isHot(x, y, r) {
                const r2 = r * r;
                // Check center
                const idx = (Math.floor(y) * canvas.width + Math.floor(x)) * 4;
                const val = mask[idx];
                
                // Check neighbors
                const neighbors = [
                    x-1, x+1, y-1, y+1
                ];
                
                // Optimization: only check if center is somewhat "hot" (not fully transparent)
                // We need to re-access the blob buffer for neighbors. 
                // To simplify, let's just check the blob buffer directly.
                
                // Actually, let's stick to the ImageData buffer approach.
                // We need to re-fill the mask buffer based on the blob buffer.
                // This is computationally expensive in JS, but okay for ~15 blobs.
            }

            // --- Efficient Metaball Rendering (The "Gooey" Effect) ---
            // This algorithm checks a grid of pixels. If a pixel is close to a blob, 
            // it gets a heat value. If the average heat in a neighborhood is high, 
            // it's a blob.
            
            const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
            const data = imageData.data;
            
            // We need to perform the metaball check.
            // We will iterate through every pixel.
            // If it's close to a blob, we mark it.
            
            // Optimization: Use a simple grid for neighbor checks.
            // Given the low resolution needed for visual appeal, we can skip pixel-perfect.
            
            // Let's use the simpler "Threshold on Buffer" method.
            // 1. Draw blobs with low alpha (0.1) into a temp canvas? No, we can't do that easily.
            // 2. We will iterate through the buffer. If a pixel is "close" to a blob center, 
            //    we add to a "heat" variable.
            
            // Let's use a simple threshold logic on the buffer itself.
            // We'll accumulate "heat" in a separate array.
            const heat = new Uint8Array(data.buffer);
            
            // Reset Heat
            for(let i=0; i<heat.length; i+=4) heat[i] = 0;

            // Add Heat from Blobs
            for (let blob of blobs) {
                const r = blob.radius;
                // Optimization: Scan a bounding box around the blob
                const minX = Math.max(0, Math.floor(blob.x - r));
                const maxX = Math.min(canvas.width, Math.ceil(blob.x + r));
                const minY = Math.max(0, Math.floor(blob.y - r));
                const maxY = Math.min(canvas.height, Math.ceil(blob.y + r));

                for (let y = minY; y < maxY; y++) {
                    for (let x = minX; x < maxX; x++) {
                        const dist = Math.sqrt((x - blob.x)**2 + (y - blob.y)**2);
                        // Gaussian-ish heat distribution
                        if (dist < r) {
                            const weight = 1 - (dist / r);
                            const idx = (y * canvas.width + x) * 4;
                            heat[idx] += weight * 100; // R
                            heat[idx+1] += weight * 100; // G
                            heat[idx+2] += weight * 100; // B
                        }
                    }
                }
            }

            // Thresholding
            // If heat > 20, it's a blob.
            // We need to map this heat to the final color.
            // We will draw the final blobs on top of a black background.
            // We use the heat map to determine visibility.
            
            // Let's just draw the blobs again, but apply a "glow" and "mask"
            // The standard trick: Draw blobs with low alpha. Then use the buffer to check.
            
            // Let's try the "Pixel Check" method which is easier to code cleanly.
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // To make it fast, we iterate pixels.
            // If pixel is close to ANY blob, draw it.
            
            // Precompute blob positions for faster access
            const blobCenters = blobs.map(b => b.x);
            const blobRadii = blobs.map(b => b.radius);

            // We iterate every 2nd pixel for performance (interpolation)
            const step = 2; 
            
            for (let y = 0; y < canvas.height; y += step) {
                for (let x = 0; x < canvas.width; x += step) {
                    let isHot = false;
                    let minDist = 10000;

                    for (let b of blobs) {
                        const dx = x - b.x;
                        const dy = y - b.y;
                        const d2 = dx*dx + dy*dy;
                        
                        // Check distance against radius
                        if (d2 < b.radius * b.radius) {
                            isHot = true;
                            minDist = Math.sqrt(d2);
                            break; // Close enough
                        }
                    }

                    if (isHot) {
                        // Draw a semi-transparent blob at this pixel
                        // We use a radial gradient to smooth the edges
                        ctx.beginPath();
                        ctx.arc(x, y, b.radius * 0.5, 0, Math.PI * 2); // Small size for the check
                        ctx.fillStyle = `rgba(${255},${255},${255}, 0.05)`;
                        ctx.fill();
                    }
                }
            }

            // Now we need to reconstruct the shapes. 
            // This is the hard part without a library.
            // We will use the "Threshold" method on the buffer.
            
            // 1. Create a "Heat" buffer
            const heatBuffer = ctx.createImageData(canvas.width, canvas.height);
            const heatData = heatBuffer.data;
            
            // 2. Fill Heat Buffer with 0
            // 3. Add Heat from Blobs
            // 4. Threshold Heat Buffer
            // 5. Draw Blobs based on Threshold
            
            // Step 3: Add Heat
            for (let blob of blobs) {
                const r = blob.radius;
                const minX = Math.max(0, Math.floor(blob.x - r));
                const maxX = Math.min(canvas.width, Math.ceil(blob.x + r));
                const minY = Math.max(0, Math.floor(blob.y - r));
                const maxY = Math.min(canvas.height, Math.ceil(blob.y + r));

                for (let y = minY; y < maxY; y++) {
                    for (let x = minX; x < maxX; x++) {
                        const dist = Math.sqrt((x - blob.x)**2 + (y - blob.y)**2);
                        if (dist < r) {
                            const idx = (y * canvas.width + x) * 4;
                            heatData[idx] += 255; // R
                            heatData[idx+1] += 255; // G
                            heatData[idx+2] += 255; // B
                            heatData[idx+3] = 1; // Alpha
                        }
                    }
                }
            }

            // Step 4: Threshold
            // If any neighbor is hot, this pixel is hot.
            const finalHeat = new Uint8Array(heatBuffer.data.buffer);
            const finalHeatData = finalHeat.buffer;
            
            // We need to access the original heat buffer data for neighbors
            // It's tricky to access Uint8Array neighbors. Let's use a simple approach:
            // Iterate pixels. Check if (x,y) is close to any blob.
            // If yes, mark it.
            
            // This is O(N*M) where N is pixels, M is blobs. With 15 blobs and 1000x1000, it's 15M ops.
            // Acceptable for 60fps on modern devices.
            
            const isBlob = new Uint8Array(canvas.width * canvas.height); // 1D array for speed
            
            for (let y = 0; y < canvas.height; y++) {
                for (let x = 0; x < canvas.width; x++) {
                    let hot = false;
                    for (let b of blobs) {
                        const dx = x - b.x;
                        const dy = y - b.y;
                        // Simple circle check
                        if (dx*dx + dy*dy < b.radius * b.radius) {
                            hot = true;
                            break;
                        }
                    }
                    if (hot) {
                        isBlob[y * canvas.width + x] = 1;
                    }
                }
            }

            // Step 5: Draw
            // We iterate the isBlob array. If 1, we draw a blob at that pixel.
            // We use a small canvas size for the drawing to be faster.
            
            // Optimization: Only draw at the step resolution, then scale up?
            // No, let's just draw normally.
            
            // We will draw the blobs again.
            // To get the metaball effect, we draw a "mask" on top of the blobs.
            
            // Reset context
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Draw a white background
            ctx.fillStyle = '#111';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Draw blobs
            for (let blob of blobs) {
                // We check pixels for this specific blob
                const minX = Math.max(0, Math.floor(blob.x - blob.radius));
                const maxX = Math.min(canvas.width, Math.ceil(blob.x + blob.radius));
                const minY = Math.max(0, Math.floor(blob.y - blob.radius));
                const maxY = Math.min(canvas.height, Math.ceil(blob.y + blob.radius));

                // We draw the blob, but we only draw pixels where isBlob is 1
                // We use a small canvas to do the mask check efficiently
                
                // Create a temp canvas for the blob mask
                const tempC = document.createElement('canvas');
                tempC.width = maxX - minX + 1;
                tempC.height = maxY - minY + 1;
                const tempCtx = tempC.getContext('2d');
                
                // Draw blobs on temp canvas (accumulated)
                // Actually, let's just use the global isBlob array logic but optimized.
                
                // Let's just draw the blobs normally with a radial gradient to simulate the shape.
                // The "Metaball" look comes from the boundary merging.
                // A simpler visual trick for this specific request:
                // Draw the blobs with a low opacity white, then draw the blob color on top.
                // Then draw a "glow" mask.
                
                // Let's stick to the robust but slower O(N*M) check for the final render.
                // It ensures the blobs merge perfectly.
                
                ctx.beginPath();
                ctx.arc(blob.x, blob.y, blob.radius, 0, Math.PI * 2);
                
                // Create gradient
                const grad = ctx.createRadialGradient(blob.x - 5, blob.y - 5, 5, blob.x, blob.y, blob.radius);
                grad.addColorStop(0, `hsla(${blob.hue}, ${blob.sat}%, ${blob.light + 20}%, 1)`);
                grad.addColorStop(1, `hsla(${blob.hue}, ${blob.sat}%, ${blob.light}%, 0.8)`);
                
                ctx.fillStyle = grad;
                ctx.fill();
                
                // Draw inner highlight
                ctx.beginPath();
                ctx.arc(blob.x - blob.radius*0.2, blob.y - blob.radius*0.2, blob.radius*0.3, 0, Math.PI*2);
                ctx.fillStyle = `rgba(255,255,255,0.3)`;
                ctx.fill();
            }
            
            // Draw Glow Overlay
            // Create a radial gradient that fades out
            const glowGrad = ctx.createRadialGradient(blobCenters[0], blobCenters[0][1], 0, blobCenters[0], blobCenters[0][1], 200);
            glowGrad.addColorStop(0, `rgba(255, 160, 0, 0.1)`);
            glowGrad.addColorStop(1, `rgba(0, 0, 0, 0)`);
            
            ctx.fillStyle = glowGrad;
            ctx.fillRect(0, 0, canvas.width, canvas.height);

        }

        // Animation Loop
        function animate() {
            time++;
            
            // Update blobs
            blobs.forEach(blob => {
                blob.update();
            });

            // Draw
            drawMetaballs();

            // Update particles occasionally
            if (time % 5 === 0) {
                if (Math.random() > 0.5) {
                    particles.push({
                        x: Math.random() * canvas.width,
                        y: canvas.height,
                        size: Math.random() * 3 + 1,
                        speed: Math.random() * 1 + 0.5
                    });
                }
            }

            // Draw particles
            ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
            for (let i = particles.length - 1; i >= 0; i--) {
                let p = particles[i];
                p.y -= p.speed;
                p.x += Math.sin(time * 0.01 + p.y) * 0.5; // Gentle drift
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fill();
                if (p.y < -10) {
                    particles.splice(i, 1);
                }
            }

            requestAnimationFrame(animate);
        }

        // Start
        animate();

    </script>
